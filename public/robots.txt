# Anadolu Hastaneleri Grubu

User-agent: *
Allow: /

# İş başvurusu paylaşım bağlantıları: aday kişisel verisi içerir,
# arama motorlarına hiçbir koşulda düşmemeli.
# Sayfanın kendisi de noindex meta etiketi basar; bu satır, JavaScript
# çalıştırmayan tarayıcılar için ikinci katmandır.
Disallow: /basvuru-paylasim/

# Banko görselindeki QR kodun açtığı tercüman listesi: yalnızca QR ile
# ulaşılır, aramada çıkmamalı (sayfa ayrıca noindex basar).
Disallow: /tercumanlar

# Yönetim paneli
Disallow: /admin/
